  function buildAndroidChromeIntent() {
    const target = buildChromeUserTarget();
    const targetUrl = target.toString();
    const hostAndPath = target.host + target.pathname + target.search + target.hash;
    // A real href intent is required in Kakao/Daangn WebView. JavaScript location changes
    // are often treated as a same-page refresh. The HTTPS fallback keeps source=chrome
    // so the destination is never classified as embedded again.
    const intentUrl = 'intent://' + hostAndPath +
      '#Intent;scheme=https;package=com.android.chrome;action=android.intent.action.VIEW;' +
      'category=android.intent.category.BROWSABLE;S.browser_fallback_url=' +
      encodeURIComponent(targetUrl) + ';end';
    return { targetUrl, intentUrl };
  }

  function isAndroidDevice() {
    return /Android/i.test(String(navigator.userAgent || ''));
  }

  function isEmbeddedBrowser() {
    const ua = String(navigator.userAgent || '');
    const source = String(new URLSearchParams(window.location.search).get('source') || '').toLowerCase();
    // The Chrome intent target explicitly uses source=chrome. Never classify that page
    // as embedded merely because document.referrer still contains Kakao or Daangn.
    if (source === 'chrome' || source === 'pwa') return false;
    if (source === 'embedded') return true;
    return /;\s*wv\)|\bwv\b|KAKAOTALK|KakaoTalk|DaumApps|NAVER|Instagram|FBAN|FBAV|Line\/|Daangn|Danggeun|Karrot|com\.towneers|EverytimeApp|WhaleInApp/i.test(ua) ||
      !!window.NativeVoice;
  }

  function isIOSSafari() {
    const ua = String(navigator.userAgent || '');
    return iosDevice && /Safari/i.test(ua) && !/CriOS|FxiOS|EdgiOS|OPiOS|KAKAOTALK|DaumApps|NAVER|Instagram|FBAN|FBAV/i.test(ua);
  }

  function detectBrowserContext() {
    if (isStandaloneApp()) return 'standalone';
    if (androidDevice && embeddedBrowser) return 'android-embedded';
    if (androidDevice) return 'android-chrome';
    if (iosDevice && embeddedBrowser) return 'ios-embedded';
    if (isIOSSafari()) return 'ios-safari';
    if (iosDevice) return 'ios-browser';
    return 'other';
  }

  function isStandaloneApp() {
    return window.matchMedia('(display-mode: standalone)').matches ||
      window.navigator.standalone === true;
  }

  function syncStandaloneInstallState() {
    if (!isStandaloneApp()) return false;
    installAttemptPending = false;
    deferredInstallPrompt = null;
    setInstallIntentArmed(false);
    clearTimeout(installAttemptTimer);
    try {
      localStorage.setItem(STORAGE.installConfirmed, '1');
      localStorage.setItem(STORAGE.installChoice, 'on');
      sessionStorage.removeItem(STORAGE.installRecheck);
    } catch (_) {}
    return true;
  }

  function isUserAppInstalledState() {
    if (syncStandaloneInstallState()) return true;
    return localStorage.getItem(STORAGE.installConfirmed) === '1';
  }
